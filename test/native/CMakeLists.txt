cmake_minimum_required(VERSION 3.20)
project(mbnime_updater_tests LANGUAGES CXX)
add_executable(updater_test WIN32 ../../windows/updater/updater.cpp)
target_compile_features(updater_test PRIVATE cxx_std_17)
target_compile_definitions(updater_test PRIVATE UNICODE _UNICODE NOMINMAX MBNIME_UPDATER_TEST)
target_compile_options(updater_test PRIVATE /EHsc /utf-8)
target_link_libraries(updater_test PRIVATE shell32)
foreach(variant old new fail)
  add_executable(${variant}_fixture WIN32 updater_fixture.cpp)
  target_compile_features(${variant}_fixture PRIVATE cxx_std_17)
  target_compile_definitions(${variant}_fixture PRIVATE UNICODE _UNICODE FIXTURE_VERSION="${variant}")
  target_link_libraries(${variant}_fixture PRIVATE shell32)
  if(variant STREQUAL "fail")
    target_compile_definitions(${variant}_fixture PRIVATE FIXTURE_FAIL)
  endif()
  set_target_properties(${variant}_fixture PROPERTIES OUTPUT_NAME mbnime RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/${variant}")
endforeach()
